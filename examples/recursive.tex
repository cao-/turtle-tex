\load[turtle]

\def\zero{0}
\def\round#1{\_directlua {tex.print(string.format('\_pcent .\zero f',#1))}}

\def\tree#1 {\ifnum\round{#1}<5
\forward #1 \penup \back #1 \pendown
\else
\forward #1/3
\left 30 \tree #1*2/3 \right 30
\forward #1/6
\right 45 \tree #1/2 \left 45
\forward #1/3
\right 15 \tree #1/2 \left 15
\forward #1/6
\penup \back #1 \pendown
\fi}

\def\fern#1 #2 {\ifnum\round{#1}<1\else
\forward #1
\right 70*#2 \fern #1*0.5 #2*-1 \left 70*#2
\forward #1
\left 70*#2 \fern #1*0.5 #2 \right 70*#2
\right 7*#2 \fern (#1-1) #2 \left 7*#2
\penup \forward -#1*2 \pendown
\fi}
