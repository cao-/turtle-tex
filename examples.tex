Squares and rectangles
\def\squarepiece #1 {%
    \forward #1
    \right 90
}
\def\square #1 {%
    \repeat 4 {%
        \squarepiece #1
    }
}
\def\rectangle #1 #2 {%
    \repeat 2 {%
        \squarepiece #1
        \squarepiece #2
    }
}

\vskip 110pt
\turtleline{\square 100, \rectangle 100 50,}

Circles and arcs
\def\arcr #1 #2 {%
    \repeat #2 {%
        \forward #1
        \right 1 
    }
}
\def\arcl #1 #2 {%
    \repeat #2 {%
        \forward #1
        \left 1 
    }
}
\def\circles {%
    \repeat 9 {%
        \arcr 0.5 360
        \right 40
    }
}
\def\petal #1 {%
    \repeat 2 {%
        \arcr #1 60
        \right 120
    }
}
\def\flower #1 {%
    \repeat 6 {%
        \petal #1
        \right 60
    }
}
\def\ray #1 {%
    \repeat 2 {%
        \arcl #1 90
        \arcr #1 90
    }
}
\def\sun #1 {%
    \repeat 9 {%
        \ray #1
        \right 160
    }
}
\vskip 70pt
\turtleline{\circles, \flower 1, \sun 0.3,}

Some poligons
\def\poly #1 #2 #3 {%
    \tmpnum=1
    \edef\side{\expr{#1}}%
    \edef\angle{\expr{#2}}%
    \edef\n{\expr[0]{#3}}
    \def\polyiter {%
        \forward {\side} 
        \right {\angle}
        \ifnum\tmpnum<\n \advance\tmpnum by1
            \ea\polyiter
        \fi
    }
    \polyiter
}

\vskip 70pt
\turtleline{\poly 50 72 5, \poly 30 60 6, \poly 50 144 5, \poly 70 135 8, \poly 60 108 10,}
