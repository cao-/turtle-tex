Squares and rectangles
\def\squarepiece #1 {%
    \forward #1
    \right 90
}
\def\square #1 {%
    \repeat 4 {%
        \squarepiece #1
    }
}
\def\rectangle #1 #2 {%
    \repeat 2 {%
        \squarepiece #1
        \squarepiece #2
    }
}

\vskip 110pt
\line{%
    \begturtle
        \square 100
    \endturtle
    \hfill
    \begturtle
        \rectangle 100 50
    \endturtle
    \hfill
}

\def\circle #1 {%
    \repeat 360 {%
        \forward #1
        \right 1
    }
